{{ define "main" }} {{- /* Legal / license template (design: LegalPage.dc.html). Covers the license and charter documents (Artistic Licenses, CLA, CPAN guidelines, bylaws, articles, committee charters, …). CONTENT RULE: all copy stays in the Markdown source. This template never hard-codes page text — it styles the *rendered* `.Content` with light CSS and a few pattern-gated `replaceRE` transforms. Every special treatment degrades to plain prose when the pattern is absent, so pages that are not license text still render cleanly: * Notice callout — the paragraphs before the first heading, shown in the violet mono callout, but ONLY when the document opens with a copyright line (the "copyright and notice" block of a license). Other pages keep those paragraphs in the body. * Section headings — `##`/`###` become the uppercase, top-ruled section headers and fill the sticky "Sections" list (built from the rendered headings, so anchors match Hugo's auto heading ids). * Definition rows — a paragraph that opens with a quoted term (`"Package" means …`) gets the term bolded with a bottom rule. * Numbered clauses — a paragraph that opens with `(1)` / `(a)` / `(i)` gets the marker in a mono gutter (coral for top-level, grey for nested). The markdown is never rewritten; the transforms only annotate the HTML Hugo already produced. */ -}} {{- $content := .Content -}} {{- $hasHeading := findRE `]` $content 1 -}} {{- $notice := "" -}} {{- $body := $content -}} {{- if and $hasHeading (findRE `(?s)^\s*

Copyright` $content 1) -}} {{- $notice = replaceRE `(?s)((“[^<]+?”(?: and “[^<]+?”)?(?: form)?) ` `

${1} ` $body -}} {{- /* Numbered clauses (marker in a mono gutter). Levels are disjoint, so the order of these passes does not matter. */ -}} {{- $body = replaceRE `

\((\d{1,2})\) ` `

(${1}) ` $body -}} {{- $body = replaceRE `

\((i{1,3}|iv|v)\) ` `

(${1}) ` $body -}} {{- $body = replaceRE `

\(([a-c])\) ` `

(${1}) ` $body -}} {{- $heads := findRE `(?s)]*>.*?` $body -}} {{- $showToc := gt (len $heads) 1 -}}

{{- end }}
